45void iTLBDebugComponent::F0() {
46 FetchPacket packet;
47
48 if (!this->fetch->ReceiveResponse(this->fetchConnectionID, &packet)) {
49 this->fetchBuffer.Enqueue(&packet);
50 this->waitingFor -= 1;
51 SINUCA3_DEBUG_PRINTF("%p: F0: Fetched instruction %s\n", this,
52 packet.response.staticInfo->instMnemonic);
53 }
54
55 if (this->waitingFor < this->fetchBuffer.GetSize() &&
56 !this->fetchBuffer.IsFull()) {
57 packet.request = 0;
58 this->fetch->SendRequest(this->fetchConnectionID, &packet);
59 this->waitingFor += 1;
60 SINUCA3_DEBUG_PRINTF("%p: F0: Sending new fetcher request. (%d/%d)\n",
61 this, this->fetchBuffer.GetOccupation(),
62 this->fetchBuffer.GetSize());
63 } else {
64 SINUCA3_DEBUG_PRINTF(
65 "%p: F0: NOT sending new fetcher request. (%d/%d)\n", this,
66 this->fetchBuffer.GetOccupation(), this->fetchBuffer.GetSize());
67 }
68}

70void iTLBDebugComponent::F1() {
71 FetchPacket packet;
72 Address fakePhysicalAddress;
73
74 if (this->itlb->ReceiveResponse(this->itlbID, &fakePhysicalAddress)) {
75 SINUCA3_DEBUG_PRINTF("%p: F1: Waiting response from iTLB.\n", this);
76 } else {
77 this->tlbRequestBuffer.Dequeue(&fakePhysicalAddress);
78 SINUCA3_DEBUG_PRINTF("%p: F1: Response from itlb received!\n", this);
79 }
80
81 if (!this->tlbRequestBuffer.IsFull() &&
82 !this->fetchBuffer.Dequeue(&packet)) {
83 Address virtualAddress =
84 static_cast<Address>(packet.response.staticInfo->instAddress);
85 this->itlb->SendRequest(this->itlbID, &virtualAddress);
86 this->tlbRequestBuffer.Enqueue(&virtualAddress);
87 SINUCA3_DEBUG_PRINTF("%p: F1: Sending request %p to itlb.\n", this,
88 (void*)virtualAddress);
89 } else {
90 if (this->tlbRequestBuffer.IsFull()) {
91 SINUCA3_DEBUG_PRINTF(
92 "%p: F1: NOT sending request to itlb. Waiting response.\n",
93 this);
94 } else {
95 SINUCA3_DEBUG_PRINTF(
96 "%p: F1: NOT sending request to itlb. Cant dequeue new "
97 "instruction.\n",
98 this);
99 }
100 }
