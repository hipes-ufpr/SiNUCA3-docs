63 if (config.ComponentReference("sendTo", &this->sendTo, true)) return 1;

71template <typename Type>
72void Queue<Type>::Clock() {
73 long numberOfConnections = this->GetNumberOfConnections();
74 Type packet;
75
76 for (long i = 0; i < numberOfConnections; ++i) {
77 while (this->ReceiveRequestFromConnection(i, &packet) == 0) {
78 if (this->sendTo->SendRequest(this->connectionID, &packet) != 0) {
79 return;
80 }
81 }
82 }
