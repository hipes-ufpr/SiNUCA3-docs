31int NewComponentDefinition(Map<Definition>* definitions,
32 Map<Linkable*>* aliases,
33 std::vector<InstanceWithDefinition>* instances,
34 Map<yaml::YamlValue>* config, const char* name,
35 const char* alias, yaml::YamlLocation location) {
36 yaml::YamlValue* clazzYaml = config->Get("class");
37
38 if (clazzYaml == NULL) {
39 SINUCA3_ERROR_PRINTF("%s:%lu:%lu Component class not passed.\n",
40 location.file, location.line, location.column);
41 return 1;
42 }
43 if (clazzYaml->type != yaml::YamlValueTypeString) {
44 SINUCA3_ERROR_PRINTF("%s:%lu:%lu Component class is not a string.\n",
45 clazzYaml->location.file, clazzYaml->location.line,
46 clazzYaml->location.column);
47 return 1;
48 }
49 const char* clazz = clazzYaml->value.string;
50
51 Definition definition;
52 definition.config = config;
53 definition.location = location;
54 definitions->Insert(name, definition);
55
56 if (alias != NULL) {
57 Linkable* newComponent = CreateComponentByClass(clazz);
58 if (newComponent == NULL) {
59 SINUCA3_ERROR_PRINTF(
60 "%s:%lu:%lu Component class %s doesn't exists.\n",
61 clazzYaml->location.file, clazzYaml->location.line,
62 clazzYaml->location.column, clazz);
63 return 1;
64 }
65 aliases->Insert(alias, newComponent);
66
67 InstanceWithDefinition instanceWithDefinition;
68 instanceWithDefinition.definition = name;
69 instanceWithDefinition.component = newComponent;
70 instances->push_back(instanceWithDefinition);
71 }
72
73 return 0;
74}

76int Engine::Configure(Config config) {
77 std::vector<InstanceWithDefinition> instances;
78 Map<Definition>* definitions = config.Definitions();
79 Map<Linkable*>* aliases = config.Aliases();
80 // Heuristic.
81 instances.reserve(32);
82
83 aliases->Insert("ENGINE", this);
84
85 Map<yaml::YamlValue>* map = config.RawYaml();
86
87 yaml::YamlValue value;
88 map->ResetIterator();
89 for (const char* key = map->Next(&value); key != NULL;
90 key = map->Next(&value)) {
91 if (value.type != yaml::YamlValueTypeMapping) {
92 SINUCA3_ERROR_PRINTF(
93 "%s:%ld:%ld Toplevel configuration parameter is not a mapping: "
94 "%s",
95 value.location.file, value.location.line, value.location.column,
96 key);
97 return 1;
98 }
99 if (NewComponentDefinition(definitions, aliases, &instances,
100 value.value.mapping, key, value.anchor,
101 value.location) != 0) {
102 for (unsigned int i = 0; i < instances.size(); ++i)
103 delete instances[i].component;
104 return 1;
105 }
106 }
107
108 std::vector<Linkable*>* components = config.Components();
109
110 components->reserve(instances.size() + 32); // Heuristic.
111 components->push_back(this);
112 for (unsigned int i = 0; i < instances.size(); ++i)
113 components->push_back(instances[i].component);
114
115 for (unsigned int i = 0; i < instances.size(); ++i) {
116 Linkable* component = instances[i].component;
117 Definition* definition = definitions->Get(instances[i].definition);
118 assert(definition != NULL);
119 Config config = Config(components, aliases, definitions,
120 definition->config, definition->location);
121 if (component->Configure(config)) {
122 // Skip the engine.
123 for (unsigned int i = 1; i < components->size(); ++i)
124 delete (*components)[i];
125 return 1;
126 }
127 }
128
129 this->numberOfComponents = components->size();
130 this->components = new Linkable*[this->numberOfComponents];
131 for (unsigned int i = 0; i < this->numberOfComponents; ++i)
132 this->components[i] = (*components)[i];
133
134 return 0;
135}

137int Engine::SendBufferedAndFetch(int id) {
138 InstructionPacket toSend = this->fetchBuffers[id];
139 const FetchResult r = this->traceReader->Fetch(&this->fetchBuffers[id], id);
140 toSend.nextInstruction = this->fetchBuffers[id].staticInfo->instAddress;
141
142 // This unfortunately drops the packet if the buffer is full. The component
143 // must ensure the buffers never fills.
144 if (this->SendResponseToConnection(id, (FetchPacket*)&toSend) != 0) {
145 SINUCA3_WARNING_PRINTF(
146 "engine: == INSTRUCTION DROP DETECTED == core %d made requests "
147 "with a full buffer, instructions will be dropped.\n",
148 id);
149 }
150
151 if (r == FetchResultEnd) {
152 this->end = true;
153 return 1;
154 } else if (r == FetchResultError) {
155 this->error = true;
156 return 1;
157 }
158
159 ++this->fetchedInstructions;
160
161 return 0;
162}

164void Engine::Fetch(int id, FetchPacket packet) {
165 if (packet.request == 0) {
166 this->SendBufferedAndFetch(id);
167 return;
168 }
169
170 long weight = this->fetchBuffers[id].staticInfo->instSize;
171
172 while (weight < packet.request) {
173 if (this->SendBufferedAndFetch(id)) {
174 return;
175 }
176 weight += this->fetchBuffers[id].staticInfo->instSize;
177 }
178}

180void Engine::Clock() {

183
184 for (int i = 0; i < numberOfConnections; ++i) {
185 if (!this->ReceiveRequestFromConnection(i, &packet)) {
186 this->Fetch(i, packet);

205void Engine::PrintTime(time_t start, unsigned long cycle) {
206 const unsigned long remaining = this->traceSize - this->fetchedInstructions;
207
208 SINUCA3_LOG_PRINTF("engine: Heartbeat at cycle %ld.\n", cycle);
209 SINUCA3_LOG_PRINTF("engine: Remaining instructions: %ld.\n", remaining);
210
211 const time_t now = time(NULL);
212 const time_t estimatedEnd =
213 remaining == 0 ? now : (traceSize * (now - start) / remaining) + start;
214 SINUCA3_LOG_PRINTF("engine: Estimated simulation end: %s",
215 ctime(&estimatedEnd));
216}

218int Engine::SetupSimulation(TraceReader* traceReader) {
219 this->traceReader = traceReader;
220 this->numberOfFetchers = this->GetNumberOfConnections();
221 this->fetchBuffers = new InstructionPacket[this->numberOfFetchers];
222
223 // Bufferize the first instruction of each core.
224 for (long i = 0; i < this->numberOfFetchers; ++i) {
225 if (this->traceReader->Fetch(&this->fetchBuffers[i], i) !=
226 FetchResultOk) {
227 return 1;
228 }
229 ++this->fetchedInstructions;
230 }
231
232 return 0;
233}

235int Engine::Simulate(TraceReader* traceReader) {
236 if (this->SetupSimulation(traceReader)) {
237 return 1;
238 }
239
240 this->traceSize = this->GetTraceSize();
241
242 const time_t start = time(NULL);
243
244 SINUCA3_LOG_PRINTF("engine: Simulation started at %s", ctime(&start));
245 SINUCA3_LOG_PRINTF("engine: Total instructions: %ld.\n", this->traceSize);
246
247 while (!this->end && !this->error) {
248 if ((this->totalCycles + 1) % (1 << 8) == 0)
249 this->PrintTime(start, this->totalCycles + 1);
250
251 for (long i = 0; i < this->numberOfComponents; ++i)
252 this->components[i]->Clock();
253
254 for (long i = 0; i < this->numberOfComponents; ++i)
255 this->components[i]->PosClock();
256
257 ++this->totalCycles;
258 }
259
260 const time_t end = time(NULL);
261 SINUCA3_LOG_PRINTF("engine: Simulation ended at %s", ctime(&end));
262 SINUCA3_LOG_PRINTF("=== SIMULATION STATISTICS ===\n");
263
264 if (this->error) {
265 SINUCA3_ERROR_PRINTF(
266 "Simulation ended due to error in trace fetching!\n");
267 }
268
269 for (long i = 0; i < this->numberOfComponents; ++i) {
270 this->components[i]->PrintStatistics();
271 }
272
273 return this->error;
274}

276Engine::~Engine() {

278 // The first component is a pointer to the engine itself, thus we start
279 // from the second.
280 for (long i = 1; i < this->numberOfComponents; ++i) {
281 delete this->components[i];
282 }
283 delete[] this->components;
284 }

286 delete[] this->fetchBuffers;
