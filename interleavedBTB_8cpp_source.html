32int BTBEntry::Allocate(unsigned int numBanks) {
33 this->numBanks = numBanks;
34 this->targetArray = new unsigned long[numBanks];
35 if (!(this->targetArray)) {
36 SINUCA3_ERROR_PRINTF("BTB entry could not be allocated");
37 return 1;
38 }
39
40 this->branchTypes = new BranchType[numBanks];
41 if (!(this->branchTypes)) {
42 delete[] this->targetArray;
43 SINUCA3_ERROR_PRINTF("BTB entry could not be allocated");
44 return 1;
45 }
46
47 this->predictorsArray = new BimodalCounter[numBanks];
48 if (!(this->predictorsArray)) {
49 delete[] this->targetArray;
50 delete[] this->branchTypes;
51 SINUCA3_ERROR_PRINTF("BTB entry could not be allocated");
52 return 1;
53 }
54
55 for (unsigned int i = 0; i < this->numBanks; ++i) {
56 this->targetArray[i] = 0;
57 this->branchTypes[i] = BranchTypeNone;
58 }
59
60 return 0;
61}

63int BTBEntry::NewEntry(unsigned long tag, unsigned int bank,
64 unsigned long target,
65 const StaticInstructionInfo* instruction) {
66 if (bank >= this->numBanks) return 1;
67
68 this->valid = true;
69 this->entryTag = tag;
70 this->targetArray[bank] = target;
71 this->branchTypes[bank] =
72 BranchTypeFromSinucaBranch(instruction->branchType);
73
74 return 0;
75}

77int BTBEntry::UpdateEntry(unsigned long bank, bool branchState) {
78 if ((bank >= this->numBanks) || this->entryTag == 0) return 1;
79
80 this->predictorsArray[bank].UpdatePrediction(branchState);
81 return 0;

112int BranchTargetBuffer::Configure(Config config) {
113 long interleavingFactor;
114 if (config.Integer("interleavingFactor", &interleavingFactor, true))
115 return 1;
116 if (interleavingFactor <= 0)
117 return config.Error("interleavingFactor", "is not > 0.");
118 if (interleavingFactor > MAX_INTERLEAVING_FACTOR)
119 return config.Error("interleavingFactor",
120 "is > MAX_INTERLEAVING_FACTOR");
121 this->interleavingFactor = interleavingFactor;
122
123 long numberOfEntries;
124 if (config.Integer("numberOfEntries", &numberOfEntries, true)) return 1;
125 if (numberOfEntries <= 0)
126 return config.Error("numberOfEntries", "is not > 0.");
127 this->numEntries = numberOfEntries;
128
129 this->sendTo = NULL;
130 if (config.ComponentReference("sendTo", &this->sendTo)) return 1;
131 if (this->sendTo != NULL) this->sendToID = this->sendTo->Connect(0);
132
133 this->interleavingBits = floor(log2(this->interleavingFactor));
134 this->entriesBits = floor(log2(this->numEntries));
135 this->interleavingFactor = (1 << this->interleavingBits);
136 this->numEntries = (1 << this->entriesBits);
137 this->btb = new BTBEntry*[this->numEntries];
138
139 for (unsigned int i = 0; i < this->numEntries; ++i) {
140 this->btb[i] = new BTBEntry;
141 this->btb[i]->Allocate(this->interleavingFactor);
142 }
143
144 return 0;
145}

170int BranchTargetBuffer::RegisterNewBranch(
171 const StaticInstructionInfo* instruction, unsigned long target) {
172 unsigned long index = this->CalculateIndex(instruction->instAddress);
173 unsigned long tag = this->CalculateTag(instruction->instAddress);
174 unsigned int bank = this->CalculateBank(instruction->instAddress);
175
176 if (this->btb[index]->GetValid()) {
177 this->replacements++;
178 }
179
180 return this->btb[index]->NewEntry(tag, bank, target, instruction);
181}

191inline void BranchTargetBuffer::Query(const StaticInstructionInfo* instruction,
192 int connectionID) {
193 this->queries++;
194 unsigned long index = this->CalculateIndex(instruction->instAddress);
195 unsigned long tag = this->CalculateTag(instruction->instAddress);
196 BTBPacket response;
197
198 response.data.response.instruction = instruction;
199 response.data.response.target =
200 instruction->instAddress + ((this->interleavingFactor == 1)
201 ? instruction->instSize
202 : (this->interleavingFactor));
203 response.data.response.numberOfInstructions = this->interleavingFactor;
204 response.data.response.interleavingBits = this->interleavingBits;
205
206 BTBEntry* currentEntry = btb[index];
207 if (currentEntry->GetTag() == tag) {
208 this->btbHits++;
209 // BTB Hit
210 /*
211 * In a BTB hit, searches for a taken branch instruction and fills in
212 * the target address if it finds one. Marks all instructions before the
213 * taken branch as valid.
214 */
215 bool branchTaken = false;
216 response.data.response.instruction = instruction;
217 response.data.response.target =
218 instruction->instAddress + this->interleavingFactor;
219
220 // TODO: change numberOfBits field; not present in struct response
221 // response.data.response.numberOfBits = this->interleavingFactor;
222
223 for (unsigned int i = 0; i < this->interleavingFactor; ++i) {
224 if (!(branchTaken)) {
225 if ((currentEntry->GetBranchType(i) ==
226 BranchTypeUnconditional) ||
227 (currentEntry->GetPrediction(i))) {
228 response.data.response.target =
229 currentEntry->GetTargetAddress(i);
230 branchTaken = true;
231 }
232 response.data.response.validBits[i] = true;
233 } else {
234 response.data.response.validBits[i] = false;
235 }
236 }
237 response.type = BTBPacketTypeResponseBTBHit;
238 } else {
239 // BTB Miss
240 /*
241 * In a BTB Miss, it assumes that all instructions are valid and that
242 * the next fetch block is sequential.
243 */
244 response.data.response.instruction = instruction;
245 response.data.response.target =
246 instruction->instAddress + this->interleavingFactor;
247 // TODO: change numberOfBits field; not present in struct response
248 // response.data.response.numberOfBits = this->interleavingFactor;
249 for (unsigned int i = 0; i < this->interleavingFactor; ++i) {
250 response.data.response.validBits[i] = true;
251 }
252 response.type = BTBPacketTypeResponseBTBMiss;
253 }
254
255 this->SendResponseToConnection(connectionID, &response);
256 if (this->sendTo != NULL) {
257 this->sendTo->SendRequest(this->sendToID, &response);
258 }
259}

272void BranchTargetBuffer::Clock() {
273 long numberOfConnections = this->GetNumberOfConnections();
274 BTBPacket packet;
275 for (long i = 0; i < numberOfConnections; ++i) {
276 while (this->ReceiveRequestFromConnection(i, &packet) == 0) {
277 switch (packet.type) {
278 case BTBPacketTypeRequestQuery:
279 SINUCA3_DEBUG_PRINTF(
280 "[BranchTargetBuffer] %p: consulting instruction [%lx] "
281 "%s\n",
282 this, packet.data.requestQuery->instAddress,
283 packet.data.requestQuery->instMnemonic);
284 this->Query(packet.data.requestQuery, i);
285 break;
286
287 case BTBPacketTypeRequestAddEntry:
288 SINUCA3_DEBUG_PRINTF(
289 "[BranchTargetBuffer] %p: adding entry [%lx] %s\n",
290 this, packet.data.requestQuery->instAddress,
291 packet.data.requestQuery->instMnemonic);
292 this->AddEntry(packet.data.requestAddEntry.instruction,
293 packet.data.requestAddEntry.target);
294 break;
295
296 case BTBPacketTypeRequestUpdate:
297 SINUCA3_DEBUG_PRINTF(
298 "[BranchTargetBuffer] %p: updating [%lx] %s\n", this,
299 packet.data.requestQuery->instAddress,
300 packet.data.requestQuery->instMnemonic);
301 this->Update(packet.data.requestUpdate.instruction,
302 packet.data.requestUpdate.branchState);
303 break;
304
305 case BTBPacketTypeResponseBTBHit:
306 case BTBPacketTypeResponseBTBMiss:
307 SINUCA3_WARNING_PRINTF(
308 "Connection %ld send a response type message to BTB.\n",
309 i);
310 break;
311 default:
312 SINUCA3_WARNING_PRINTF(
313 "Connection %ld send a invalid message to BTB.\n", i);
314 }
315 }
316 }
317}

319void BranchTargetBuffer::PrintStatistics() {
320 for (unsigned int i = 0; i < this->numEntries; ++i) {
321 if (this->btb[i]->GetValid()) this->occupation++;
322 }
323
324 SINUCA3_LOG_PRINTF("Branch Target Buffer [%p]\n", this);
325 SINUCA3_LOG_PRINTF(" Entries: %d\n", this->numEntries);
326 SINUCA3_LOG_PRINTF(" Total Queries: %lu\n", this->queries);
327 SINUCA3_LOG_PRINTF(" Total Branches: %lu\n", this->totalBranch);
328 SINUCA3_LOG_PRINTF(" BTB Hits: %lu\n", this->btbHits);
329 SINUCA3_LOG_PRINTF(" BTB Occupation: %lu\n", this->occupation);
330 SINUCA3_LOG_PRINTF(" Entry Replacements: %lu\n", this->replacements);
331 SINUCA3_LOG_PRINTF(" Hit Ratio: %lf%%\n",
332 ((double)this->btbHits / (double)this->queries) * 100);
333 SINUCA3_LOG_PRINTF(
334 " Occupation Ratio: %lf%%\n",
335 ((double)this->occupation / (double)this->numEntries) * 100);
