66int FileHeader::FlushHeader(FILE *file) {
67 if (!file) return 1;
68 long orgPos = ftell(file);
69 rewind(file);
70 if (fwrite(this, 1, sizeof(*this), file) != sizeof(*this)) {
71 return 1;
72 }
73 fseek(file, orgPos, SEEK_SET);
74 return 0;
75}

93void FileHeader::SetHeaderType(uint8_t fileType) {
94 this->fileType = fileType;
95 if (this->fileType == FileTypeStaticTrace) {
96 strcpy((char*)this->prefix, PREFIX_STATIC_FILE);
97 } else if (this->fileType == FileTypeDynamicTrace) {
98 strcpy((char*)this->prefix, PREFIX_DYNAMIC_FILE);
99 } else if (this->fileType == FileTypeMemoryTrace) {
100 strcpy((char*)this->prefix, PREFIX_MEMORY_FILE);
101 } else {
102 SINUCA3_ERROR_PRINTF("[FileHeader] Unkown file type!\n");
103 }
