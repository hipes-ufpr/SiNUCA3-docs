34int DynamicTraceWriter::OpenFile(const char* sourceDir, const char* imageName,
35 int tid) {
36 unsigned long bufferSize;
37 char* path;
38
39 bufferSize = GetPathTidInSize(sourceDir, "dynamic", imageName);
40 path = (char*)alloca(bufferSize);
41 FormatPathTidIn(path, sourceDir, "dynamic", imageName, tid, bufferSize);
42 this->file = fopen(path, "wb");
43 if (this->file == NULL) {
44 SINUCA3_ERROR_PRINTF("Failed to alloc file in dynamic trace!");
45 return 1;
46 }
47 this->header.ReserveHeaderSpace(this->file);
48
49 return 0;
50}

52int DynamicTraceWriter::FlushRecordArray() {
53 if (this->file == NULL) {
54 SINUCA3_ERROR_PRINTF("File pointer is nil in mem trace!\n");
55 return 1;
56 }
57
58 unsigned long occupationInBytes =
59 this->recordArrayOccupation * sizeof(*this->recordArray);
60
61 if (fwrite(this->recordArray, 1, occupationInBytes, this->file) !=
62 occupationInBytes) {
63 SINUCA3_ERROR_PRINTF("Failed to flush memory records!\n");
64 return 1;
65 }
66
67 return 0;
68}

70int DynamicTraceWriter::CheckRecordArray() {
71 if (this->IsRecordArrayFull()) {
72 if (this->FlushRecordArray()) {
73 SINUCA3_ERROR_PRINTF("Failed to flush mem record array!\n")
74 return 1;
75 }
76 this->ResetRecordArray();
77 }
78 return 0;
79}

81int DynamicTraceWriter::AddDynamicRecord(DynamicTraceRecord record) {
82 this->recordArray[this->recordArrayOccupation] = record;
83 ++this->recordArrayOccupation;
84 if (this->CheckRecordArray()) return 1;
