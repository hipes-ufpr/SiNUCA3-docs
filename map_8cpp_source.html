29unsigned int map::Hash(const unsigned char* const buffer, unsigned long size) {
30 if (size <= 0) return 0;
31
32 unsigned int s = buffer[0];
33 unsigned int p = map::P;
34
35 for (unsigned long i = 1; i < size; ++i) {
36 s += buffer[i] * p;
37 p *= map::P;
38 }
39
40 return s % map::M;
41}

43const char* StringMap::Insert(const char* key, const char* value) {
44 unsigned long keyLen = strlen(key) * sizeof(char);
45 unsigned long valueLen = strlen(value) * sizeof(char);
46 unsigned int pos = map::Hash((const unsigned char* const)key, keyLen);
47
48 map::Node<char*>** nodePtr = &this->table[pos];
49 while (*nodePtr != NULL) nodePtr = &(*nodePtr)->next;
50
51 *nodePtr = (map::Node<char*>*)this->arena->Alloc(sizeof(**nodePtr));
52 (*nodePtr)->key = (char*)this->arena->Alloc(keyLen);
53 (*nodePtr)->value = (char*)this->arena->Alloc(valueLen);
54 (*nodePtr)->next = NULL;
55
56 memcpy((void*)(*nodePtr)->key, key, keyLen + 1);
57 memcpy((void*)(*nodePtr)->value, value, valueLen + 1);
58
59 return (*nodePtr)->value;
60}

62const char* StringMap::Get(const char* const key) {
63 unsigned int pos = map::Hash((const unsigned char* const)key, strlen(key));
64
65 for (map::Node<char*>* node = this->table[pos]; node != NULL;
66 node = node->next) {
67 if (strcmp(node->key, key) == 0) return node->value;
68 }
69
70 return NULL;
71}

73const char* StringMap::Next(char** elementRet) {
74 if (this->iteratorPtr != NULL) {
75 if (this->iteratorPtr->next != NULL) {
76 this->iteratorPtr = this->iteratorPtr->next;
77 *elementRet = this->iteratorPtr->value;
78 return this->iteratorPtr->key;
79 } else {
80 this->iteratorIdx += 1;
81 if (this->iteratorIdx >= map::M) {
82 this->ResetIterator();
83 return NULL;
84 }
85 }
86 }
87
88 for (; this->iteratorIdx < map::M; ++this->iteratorIdx) {
89 if (this->table[this->iteratorIdx] != NULL) {
90 this->iteratorPtr = this->table[this->iteratorIdx];
91 *elementRet = this->iteratorPtr->value;
92 return this->iteratorPtr->key;
93 }
94 }
95
96 this->ResetIterator();
97 return NULL;
98}
