33void SimpleInstructionMemory::Clock() {
34 long numberOfConnections = this->GetNumberOfConnections();
35 InstructionPacket packet;
36 for (long i = 0; i < numberOfConnections; ++i) {
37 while (this->ReceiveRequestFromConnection(i, &packet) == 0) {
38 ++this->numberOfRequests;
39 if (this->sendTo != NULL) {
40 this->sendTo->SendRequest(this->sendToID, &packet);
41 } else {
42 this->SendResponseToConnection(i, &packet);
43 }
44 }
45 }
46}
