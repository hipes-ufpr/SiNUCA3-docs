28void TraceDumperComponent::Override(const char* instruction) {
29 int size = strlen(instruction) + 1;
30 const char* str = new char[size];
31 memcpy((void*)str, (void*)instruction, size);
32 this->overrides.push_back(str);
33}

35int TraceDumperComponent::Configure(Config config) {
36 if (config.ComponentReference("fetch", &this->fetch, true)) return 1;
37 if (config.Bool("default", &this->def)) return 1;
38
39 this->fetchID = this->fetch->Connect(0);
40
41 // Use raw yaml to get the overrides.
42 yaml::YamlValue value;
43 Map<yaml::YamlValue>* rawConfig = config.RawYaml();
44 rawConfig->ResetIterator();
45 for (const char* key = rawConfig->Next(&value); key != NULL;
46 key = rawConfig->Next(&value)) {
47 if (strcmp(key, "fetch") && strcmp(key, "default")) this->Override(key);
48 }
49
50 return 0;
51}

53bool TraceDumperComponent::IsOverride(const char* instruction) {
54 for (unsigned int i = 0; i < this->overrides.size(); ++i) {
55 if (strcmp(instruction, this->overrides[i]) == 0) return true;
56 }
57 return false;
58}

60void TraceDumperComponent::Clock() {
61 FetchPacket fetch;
62 fetch.request = 0;
63 this->fetch->SendRequest(this->fetchID, &fetch);
64 if (this->fetch->ReceiveResponse(this->fetchID, &fetch) == 0) {
65 InstructionPacket instruction = fetch.response;
66 if (this->def ^ this->IsOverride(instruction.staticInfo->instMnemonic)) {
67 ++this->fetched;
68 SINUCA3_LOG_PRINTF("TraceDumperComponent %p: Fetched {\n", this);
69 SINUCA3_LOG_PRINTF(" instMnemonic: %s\n",
70 instruction.staticInfo->instMnemonic);
71 SINUCA3_LOG_PRINTF(" instAddress: %ld\n",
72 instruction.staticInfo->instAddress);
73 SINUCA3_LOG_PRINTF(" instSize: %lu\n",
74 instruction.staticInfo->instSize);
75 SINUCA3_LOG_PRINTF(" readRegs: [");
76 for (unsigned char i = 0; i < instruction.staticInfo->numberOfReadRegs;
77 ++i) {
78 SINUCA3_LOG_PRINTF("%u", instruction.staticInfo->readRegsArray[i]);
79 if (i + 1 < instruction.staticInfo->numberOfReadRegs)
80 SINUCA3_LOG_PRINTF(", ");
81 }
82 SINUCA3_LOG_PRINTF("]\n");
83 SINUCA3_LOG_PRINTF(" writeRegs: [");
84 for (unsigned char i = 0; i < instruction.staticInfo->numberOfWriteRegs;
85 ++i) {
86 SINUCA3_LOG_PRINTF("%u", instruction.staticInfo->writtenRegsArray[i]);
87 if (i + 1 < instruction.staticInfo->numberOfWriteRegs)
88 SINUCA3_LOG_PRINTF(", ");
89 }
90 SINUCA3_LOG_PRINTF("]\n");
91 SINUCA3_LOG_PRINTF(
92 " branchType: %d\n",
93 static_cast<int>(instruction.staticInfo->branchType));
94 SINUCA3_LOG_PRINTF(
95 " isIndirect: %s\n",
96 instruction.staticInfo->isIndirectControlFlowInst ? "true" : "false");
97 SINUCA3_LOG_PRINTF(
98 " isPredicated: %s\n",
99 instruction.staticInfo->isPredicatedInst ? "true" : "false");
100 SINUCA3_LOG_PRINTF(
101 " isPrefetch: %s\n",
102 instruction.staticInfo->isPrefetchHintInst ? "true" : "false");
103 SINUCA3_LOG_PRINTF("}\n");
104 }
105 }
106}
