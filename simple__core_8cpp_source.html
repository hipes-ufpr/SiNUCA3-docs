30int SimpleCore::Configure(Config config) {
31 if (config.ComponentReference("instructionMemory",
32 &this->instructionMemory))
33 return 1;
34 if (config.ComponentReference("dataMemory", &this->dataMemory)) return 1;
35 if (config.ComponentReference("fetching", &this->fetching, true)) return 1;
36
37 if (this->instructionMemory != NULL)
38 this->instructionConnectionID = this->instructionMemory->Connect(0);
39 if (this->dataMemory != NULL)
40 this->dataConnectionID = this->dataMemory->Connect(0);
41 this->fetchingConnectionID = this->fetching->Connect(0);
42
43 return 0;
44}

46void SimpleCore::Clock() {
47 FetchPacket fetch;
48 fetch.request = 0;
49 this->fetching->SendRequest(this->fetchingConnectionID, &fetch);
50 if (this->fetching->ReceiveResponse(this->fetchingConnectionID, &fetch) ==
51 0) {
52 ++this->numFetchedInstructions;
53 if (this->instructionMemory != NULL) {
54 MemoryPacket fetchPacket = fetch.response.staticInfo->instAddress;
55 this->instructionMemory->SendRequest(this->instructionConnectionID,
56 &fetchPacket);
57 if (this->dataMemory != NULL) {
58 for (long i = 0; i < fetch.response.dynamicInfo.numReadings;
59 ++i) {
60 this->dataMemory->SendRequest(
61 this->dataConnectionID,
62 (unsigned long*)&(
63 fetch.response.dynamicInfo.readsAddr[i]));
64 }
65
66 for (long i = 0; i < fetch.response.dynamicInfo.numWritings;
67 ++i) {
68 this->dataMemory->SendRequest(
69 this->dataConnectionID,
70 (unsigned long*)&(
71 fetch.response.dynamicInfo.writesAddr[i]));
72 }
73 }
74 }
75 }
