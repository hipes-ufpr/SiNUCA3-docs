36int GsharePredictor::Allocate() {
37 this->entries = new BimodalCounter[this->numberOfEntries];
38 if (!this->entries) {
39 SINUCA3_ERROR_PRINTF("Gshare failed to allocate table\n");
40 return 1;
41 }
42 this->indexQueue.Allocate(this->indexQueueSize, sizeof(this->currentIndex));
43 return 0;
44}
45
46void GsharePredictor::Deallocate() {
47 if (this->entries) {
48 delete[] this->entries;
49 this->entries = NULL;
50 }
51 if (!this->indexQueue.IsEmpty()) {
52 SINUCA3_WARNING_PRINTF(
53 "Gshare index queue not empty when it was expected to be\n");
54 }
55 this->indexQueue.Deallocate();
56}

58int GsharePredictor::RoundNumberOfEntries(unsigned long requestedSize) {
59 unsigned int bits = (unsigned int)floor(log2(requestedSize));
60 if (bits == 0) {
61 return 1;
62 }
63 this->numberOfEntries = 1 << bits;
64 this->indexBitsSize = bits;
65 return 0;
66}

88void GsharePredictor::UpdateEntry() {
89 bool pred = this->entries[this->currentIndex].GetPrediction();
90 if (pred != this->wasBranchTaken) {
91 this->numberOfWrongPredictions++;
92 }
93 this->entries[this->currentIndex].UpdatePrediction(this->wasBranchTaken);
94}

96void GsharePredictor::UpdateGlobBranchHistReg() {
97 SINUCA3_DEBUG_PRINTF("Gshare Gbhr Bef [%ld]\n", this->globalBranchHistReg);
98 this->globalBranchHistReg <<= 1;
99 if (this->wasBranchTaken) {
100 this->globalBranchHistReg |= 1;
101 }
102 SINUCA3_DEBUG_PRINTF("Gshare Gbhr Aft [%ld]\n", this->globalBranchHistReg);
103}

105void GsharePredictor::Update() {
106 if (this->DequeueIndex()) {
107 SINUCA3_ERROR_PRINTF("Gshare table was not updated\n");
108 return;
109 }
110 this->UpdateEntry();
111 this->UpdateGlobBranchHistReg();
112}

114void GsharePredictor::QueryEntry() {
115 this->numberOfPredictions++;
116 this->wasPredictedToBeTaken =
117 this->entries[this->currentIndex].GetPrediction();
118}

120void GsharePredictor::Query(PredictorPacket* pkt, unsigned long addr) {
121 this->CalculateIndex(addr);
122 this->QueryEntry();
123 this->PreparePacket(pkt);
124 if (this->EnqueueIndex()) {
125 SINUCA3_WARNING_PRINTF("Gshare index queue full\n");
126 }
127}

129void GsharePredictor::CalculateIndex(unsigned long addr) {
130 unsigned long mask = (1 << this->indexBitsSize) - 1;
131 this->currentIndex = (this->globalBranchHistReg ^ addr) & mask;
132 SINUCA3_DEBUG_PRINTF("Gshare Idx [%ld]\n", this->currentIndex);
133}

135int GsharePredictor::Configure(Config config) {
136 long numberOfEntries = 0;
137 if (config.Integer("numberOfEntries", &numberOfEntries, true)) return 1;
138 if (numberOfEntries <= 0)
139 return config.Error("numberOfEntries", "is not > 0.");
140 this->numberOfEntries = numberOfEntries;
141
142 long indexQueueSize = 1;
143 if (config.Integer("indexQueueSize", &indexQueueSize)) return 1;
144 if (indexQueueSize <= 0)
145 return config.Error("indexQueueSize", "is not > 0.");
146 this->indexQueueSize = indexQueueSize;
147
148 if (config.ComponentReference("sendTo", &this->sendTo)) return 1;
149 if (this->sendTo != NULL) this->sendToId = this->sendTo->Connect(0);
150
151 return this->Allocate();
152}

154void GsharePredictor::PrintStatistics() {
155 double fraction =
156 ((double)this->numberOfWrongPredictions / this->numberOfPredictions);
157 SINUCA3_LOG_PRINTF("Gshare table size [%lu] & number of index bits [%u]\n",
158 this->numberOfEntries, this->indexBitsSize);
159 SINUCA3_LOG_PRINTF("Gshare number of predictions [%lu]\n",
160 this->numberOfPredictions);
161 SINUCA3_LOG_PRINTF("Gshare number of wrong predictions [%lu]\n",
162 this->numberOfWrongPredictions);
163 SINUCA3_LOG_PRINTF("Gshare rate of wrong predictions [%.0lf]%%\n",
164 fraction * 100);
165}

167void GsharePredictor::Clock() {
168 PredictorPacket packet;
169 unsigned long addr;
170 long totalConnections = this->GetNumberOfConnections();
171 for (long i = 0; i < totalConnections; i++) {
172 while (this->ReceiveRequestFromConnection(i, &packet) == 0) {
173 if (packet.type == PredictorPacketTypeRequestQuery) {
174 addr = packet.data.requestQuery.staticInfo->instAddress;
175 this->Query(&packet, addr);
176 if (this->sendTo == NULL) {
177 this->SendResponseToConnection(i, &packet);
178 } else {
179 this->sendTo->SendRequest(sendToId, &packet);
180 }
181 }
182 if (packet.type == PredictorPacketTypeRequestDirectionUpdate) {
183 this->wasBranchTaken = packet.data.directionUpdate.taken;
184 this->Update();
185 }
186 }
187 }
188}
