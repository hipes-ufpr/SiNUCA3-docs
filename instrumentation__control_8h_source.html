79#define DEFINE_INTRINSIC(name) \
80 __asm__( \
81 ".intel_syntax noprefix\n" \
82 ".section .text\n" \
83 ".type __" #name \
84 "Loader,@function\n" \
85 "__" #name \
86 "Loader:\n" \
87 "\tpush rbp\n" \
88 "\tmov rbp, rsp\n" \
89 "\tpush rax\n" /* We will need to keep it's value. */ \
90 "\tlea rax, [rip + " #name \
91 "]\n" \
92 "\tmov [rip + __intrinsicCall], rax\n" \
93 "\tpop rax\n" /* Restore rax value to perform the context switch. */ \
94 "\tcall __IntrinsicsSwitchContext\n" \
95 "\tpop rbp\n" \
96 "\tret\n" \
97 ".att_syntax prefix\n"); \
