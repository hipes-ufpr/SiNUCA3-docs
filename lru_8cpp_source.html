32LRU::LRU(int numSets, int numWays) : ReplacementPolicy(numSets, numWays) {
33 this->WayUsageCounters = new unsigned int*[this->numSets];
34 int n = this->numSets * this->numWays;
35 this->WayUsageCounters[0] = new unsigned int[n];
36 memset(this->WayUsageCounters[0], 0, n * sizeof(unsigned int));
37 for (int i = 1; i < this->numSets; i++) {
38 this->WayUsageCounters[i] =
39 this->WayUsageCounters[0] + (i * this->numWays);
40 }
41};
42
43LRU::~LRU() {
44 delete[] this->WayUsageCounters[0];
45 delete[] this->WayUsageCounters;
46}

48void LRU::Acess(CacheLine* entry) {
49 for (int way = 0; way < this->numWays; ++way) {
50 this->WayUsageCounters[entry->i][way] += 1;
51 }
52 this->WayUsageCounters[entry->i][entry->j] = 0;
53}

55void LRU::SelectVictim(unsigned long tag, unsigned long index, int* resultSet,
56 int* resultWay) {
57 (void)tag;
58 long lruMaxCount = -1; // Max value
59 *resultSet = index;
60 for (int way = 0; way < this->numWays; ++way) {
61 if (this->WayUsageCounters[index][way] > lruMaxCount) {
62 lruMaxCount = this->WayUsageCounters[index][way];
63 *resultWay = way;
64 }
65 }
66}
