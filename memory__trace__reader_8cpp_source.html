32int MemoryTraceReader::OpenFile(const char* sourceDir, const char* imageName,
33 int tid) {
34 unsigned long bufferSize;
35 char* path;
36
37 bufferSize = GetPathTidInSize(sourceDir, "memory", imageName);
38 path = (char*)alloca(bufferSize);
39 FormatPathTidIn(path, sourceDir, "memory", imageName, tid, bufferSize);
40 this->file = fopen(path, "rb");
41 if (this->file == NULL) {
42 SINUCA3_ERROR_PRINTF("Failed to open memory trace file!\n");
43 return 1;
44 }
45 if (this->header.LoadHeader(this->file)) {
46 SINUCA3_ERROR_PRINTF("Failed to read memory trace header!\n");
47 return 1;
48 }
49
50 return 0;
51}

53int MemoryTraceReader::ReadMemoryOperations(InstructionPacket* inst) {
54 if (this->reachedEnd) {
55 SINUCA3_ERROR_PRINTF(
56 "[ReadMemoryOperations] already reached end in mem trace file!\n");
57 return 1;
58 }
59
60 if (this->recordArrayIndex == this->numberOfRecordsRead) {
61 if (this->LoadRecordArray()) {
62 this->reachedEnd = true;
63 return 1;
64 }
65 }
66
67 if (this->recordArray[this->recordArrayIndex].recordType !=
68 MemoryRecordHeader) {
69 SINUCA3_ERROR_PRINTF(
70 "[ReadMemoryOperations] Expected memory operation header!\n");
71 SINUCA3_ERROR_PRINTF(
72 "[ReadMemoryOperations] recordType is [%d] and record array index "
73 "is [%d]\n",
74 this->recordArray[this->recordArrayIndex].recordType,
75 this->recordArrayIndex);
76 return 1;
77 }
78
79 int totalMemOps =
80 this->recordArray[this->recordArrayIndex].data.numberOfMemoryOps;
81
82 ++this->recordArrayIndex;
83
84 unsigned char recordType;
85 for (int i = 0; i < totalMemOps; ++i, ++this->recordArrayIndex) {
86 if (this->recordArrayIndex == this->numberOfRecordsRead) {
87 if (this->LoadRecordArray()) {
88 SINUCA3_ERROR_PRINTF(
89 "[ReadMemoryOperations] invalid number of mem ops!\n");
90 this->reachedEnd = true;
91 return 1;
92 }
93 }
94
95 recordType = this->recordArray[this->recordArrayIndex].recordType;
96 if (recordType == MemoryRecordLoad) {
97 inst->dynamicInfo.readsAddr[inst->dynamicInfo.numReadings] =
98 this->recordArray[this->recordArrayIndex]
99 .data.operation.address;
100 inst->dynamicInfo.readsSize[inst->dynamicInfo.numReadings] =
101 this->recordArray[this->recordArrayIndex].data.operation.size;
102 inst->dynamicInfo.numReadings++;
103 } else if (recordType == MemoryRecordStore) {
104 inst->dynamicInfo.writesAddr[inst->dynamicInfo.numWritings] =
105 this->recordArray[this->recordArrayIndex]
106 .data.operation.address;
107 inst->dynamicInfo.writesSize[inst->dynamicInfo.numWritings] =
108 this->recordArray[this->recordArrayIndex].data.operation.size;
109 inst->dynamicInfo.numWritings++;
110 } else {
111 SINUCA3_ERROR_PRINTF(
112 "[ReadMemoryOperations] unexpected record type!\n");
113 return 1;
114 }
115 }
116
117 return 0;
118}

120int MemoryTraceReader::LoadRecordArray() {
121 this->recordArrayIndex = 0;
122
123 unsigned long readBytes = 0;
124 readBytes =
125 fread(this->recordArray, 1, sizeof(this->recordArray), this->file);
126
127 this->numberOfRecordsRead = readBytes / sizeof(*this->recordArray);
128
129 return (this->numberOfRecordsRead == 0);
130}
