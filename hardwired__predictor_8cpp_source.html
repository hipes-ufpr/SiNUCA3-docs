32int HardwiredPredictor::Configure(Config config) {
33 if (config.Bool("syscall", &this->syscall)) return 1;
34 if (config.Bool("call", &this->call)) return 1;
35 if (config.Bool("return", &this->ret)) return 1;
36 if (config.Bool("uncond", &this->uncond)) return 1;
37 if (config.Bool("cond", &this->cond)) return 1;
38 if (config.Bool("noBranch", &this->noBranch)) return 1;
39
40 if (config.ComponentReference("sendTo", &this->sendTo)) return 1;
41 if (this->sendTo != NULL) this->sendToID = this->sendTo->Connect(0);
42
43 return 0;
44}

46void HardwiredPredictor::Respond(int id, PredictorPacket request) {
47 if (request.type == PredictorPacketTypeRequestTargetUpdate ||
48 request.type == PredictorPacketTypeRequestDirectionUpdate)
49 return;
50 const InstructionPacket instruction = request.data.requestQuery;
51 bool predict = true;
52
53 // TODO: check if this is the right fix
54 if (instruction.staticInfo->branchType != BranchNone) {
55 ++this->numberOfNoBranchs;
56 predict = this->noBranch;
57 } else {
58 switch (instruction.staticInfo->branchType) {
59 case BranchSyscall:
60 predict = this->syscall;
61 ++this->numberOfSyscalls;
62 break;
63 case BranchCall:
64 predict = this->call;
65 ++this->numberOfCalls;
66 break;
67 case BranchRet:
68 predict = this->ret;
69 ++this->numberOfRets;
70 break;
71 case BranchUncond:
72 predict = this->uncond;
73 ++this->numberOfUnconds;
74 break;
75 case BranchCond:
76 predict = this->cond;
77 ++this->numberOfConds;
78 break;
79 case BranchSysret:
80 predict = this->sysret;
81 ++this->numberOfSysrets;
82 break;
83 case BranchNone:
84 break;
85 }
86 }
87
88 PredictorPacket response;
89 response.type = PredictorPacketTypeResponseTakeToAddress;
90 response.data.targetResponse.instruction = instruction;
91 if (predict) {
92 response.data.targetResponse.target = instruction.nextInstruction;
93 } else {
94 // Fast way to create an address that isn't nextInstruction.
95 response.data.targetResponse.target = ~instruction.nextInstruction;
96 }
97
98 this->SendResponseToConnection(id, &response);
99 if (this->sendTo != NULL) {
100 this->sendTo->SendRequest(this->sendToID, &response);
101 }
102}

104void HardwiredPredictor::Clock() {
105 const unsigned long numberOfConnections = this->GetNumberOfConnections();
106 for (unsigned long i = 0; i < numberOfConnections; ++i) {
107 PredictorPacket packet;
108 while (this->ReceiveRequestFromConnection(i, &packet) == 0) {
109 this->Respond(i, packet);
110 }
111 }
112}
