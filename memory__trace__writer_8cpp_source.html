32int MemoryTraceWriter::OpenFile(const char* sourceDir, const char* imageName,
33 int tid) {
34 unsigned long bufferSize;
35
36 bufferSize = GetPathTidInSize(sourceDir, "memory", imageName);
37 char* path = (char*)alloca(bufferSize);
38 FormatPathTidIn(path, sourceDir, "memory", imageName, tid, bufferSize);
39 this->file = fopen(path, "wb");
40 if (this->file == NULL) {
41 SINUCA3_ERROR_PRINTF("Failed to alloc this->file\n");
42 return 1;
43 }
44 this->header.ReserveHeaderSpace(this->file);
45
46 return 0;
47}

49int MemoryTraceWriter::FlushRecordArray() {
50 if (this->file == NULL) {
51 SINUCA3_ERROR_PRINTF("File pointer is nil in mem trace!\n");
52 return 1;
53 }
54
55 unsigned long occupationInBytes =
56 this->recordArrayOccupation * sizeof(*this->recordArray);
57
58 if (fwrite(this->recordArray, 1, occupationInBytes, this->file) !=
59 occupationInBytes) {
60 SINUCA3_ERROR_PRINTF("[1] Failed to flush memory records!\n");
61 return 1;
62 }
63
64 return 0;
65}

67int MemoryTraceWriter::CheckRecordArray() {
68 if (this->IsRecordArrayFull()) {
69 if (this->FlushRecordArray()) {
70 SINUCA3_ERROR_PRINTF("Failed to flush mem record array!\n")
71 return 1;
72 }
73 this->ResetRecordArray();
74 }
75 return 0;
76}

78int MemoryTraceWriter::AddMemoryRecord(MemoryTraceRecord record) {
79 this->recordArray[this->recordArrayOccupation] = record;
80 ++this->recordArrayOccupation;
81 if (this->CheckRecordArray()) return 1;
