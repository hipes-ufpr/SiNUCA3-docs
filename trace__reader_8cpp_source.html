29int SinucaTraceReader::OpenTrace(const char *imageName, const char *sourceDir) {
30 this->staticTrace = new StaticTraceReader;
31 if (this->staticTrace == NULL) {
32 SINUCA3_ERROR_PRINTF("Failed to create static trace reader\n");
33 return 1;
34 }
35 if (this->staticTrace->OpenFile(sourceDir, imageName)) {
36 SINUCA3_ERROR_PRINTF("Failed to open static trace\n");
37 return 1;
38 }
39
40 this->totalThreads = this->staticTrace->GetNumThreads();
41 this->totalBasicBlocks = this->staticTrace->GetTotalBasicBlocks();
42 this->totalStaticInst = this->staticTrace->GetTotalInstInStaticTrace();
43 this->traceFilesVersion = this->staticTrace->GetVersionInt();
44 this->traceFilesTargetArch = this->staticTrace->GetTargetInt();
45
46 SINUCA3_WARNING_PRINTF("Trace files:\n");
47 SINUCA3_WARNING_PRINTF("\t Version: %d\n", this->traceFilesVersion);
48 SINUCA3_WARNING_PRINTF("\t Target: %s\n", staticTrace->GetTargetString());
49
50 for (int i = 0; i < this->totalThreads; ++i) {
51 ThreadData *tData = new ThreadData;
52 if (tData == NULL) {
53 SINUCA3_ERROR_PRINTF("[OpenTrace] failed to alloc ThreadData!\n");
54 return 1;
55 }
56
57 this->threadDataVec.push_back(tData);
58
59 if (tData->Allocate(sourceDir, imageName, i)) {
60 SINUCA3_ERROR_PRINTF("[OpenTrace] tData Allocate method failed!\n");
61 return 1;
62 }
63 if (tData->CheckVersion(this->traceFilesVersion)) {
64 SINUCA3_ERROR_PRINTF("[OpenTrace] incompatible version!\n");
65 return 1;
66 }
67 if (tData->CheckTargetArch(this->traceFilesTargetArch)) {
68 SINUCA3_ERROR_PRINTF("[OpenTrace] incompatible target!\n");
69 return 1;
70 }
71 }
72
73 this->reachedAbruptEnd = false;
74
75 if (this->GenerateInstructionDict()) {
76 SINUCA3_ERROR_PRINTF("[OpenTrace] Failed to generate instruction "
77 "dictionary\n");
78 return 1;
79 }
80
81 return 0;
82}

84int SinucaTraceReader::GenerateInstructionDict() {
85 StaticInstructionInfo *instInfoPtr;
86 unsigned long poolOffset;
87 unsigned long bblCounter;
88 unsigned int bblSize;
89 unsigned int instCounter;
90 StaticTraceRecordType recordType;
91
92 this->basicBlockSizeArr = new int[this->totalBasicBlocks];
93 if (this->basicBlockSizeArr == NULL) {
94 SINUCA3_ERROR_PRINTF("Failed to alloc basicBlockSizeArr\n");
95 return 1;
96 }
97
98 this->instructionDict = new StaticInstructionInfo *[this->totalBasicBlocks];
99 if (this->instructionDict == NULL) {
100 SINUCA3_ERROR_PRINTF("Failed to alloc instructionDict\n");
101 return 1;
102 }
103
104 this->instructionPool = new StaticInstructionInfo[this->totalStaticInst];
105 if (this->instructionPool == NULL) {
106 SINUCA3_ERROR_PRINTF("Failed to alloc instructionPool\n");
107 return 1;
108 }
109
110 poolOffset = 0;
111
112 for (bblCounter = 0; bblCounter < this->totalBasicBlocks; bblCounter++) {
113 if (this->staticTrace->ReadStaticRecordFromFile()) {
114 return 1;
115 }
116
117 recordType = this->staticTrace->GetStaticRecordType();
118 if (recordType != StaticRecordBasicBlockSize) {
119 SINUCA3_ERROR_PRINTF("Expected basic block size record type\n");
120 return 1;
121 }
122
123 bblSize = this->staticTrace->GetBasicBlockSize();
124 this->basicBlockSizeArr[bblCounter] = bblSize;
125 this->instructionDict[bblCounter] = &this->instructionPool[poolOffset];
126 poolOffset += bblSize;
127
128 for (instCounter = 0; instCounter < bblSize; instCounter++) {
129 if (this->staticTrace->ReadStaticRecordFromFile()) {
130 return 1;
131 }
132
133 recordType = this->staticTrace->GetStaticRecordType();
134 if (recordType != StaticRecordInstruction) {
135 SINUCA3_ERROR_PRINTF("Expected instruction record type\n");
136 return 1;
137 }
138
139 instInfoPtr = &this->instructionDict[bblCounter][instCounter];
140 this->staticTrace->TranslateRawInstructionToSinucaInst(instInfoPtr);
141 }
142 }
143
144 return 0;
145}

147bool SinucaTraceReader::HasExecutionEnded() {
148 if (this->reachedAbruptEnd) return true;
149
150 if (this->threadDataVec[0]->dynFile.HasReachedEnd()) {
151 for (int i = 1; i < this->totalThreads; ++i) {
152 if (!this->threadDataVec[i]->dynFile.HasReachedEnd()) {
153 SINUCA3_ERROR_PRINTF("Thread [%d] file hasnt reached end!\n", i);
154 }
155 }
156 return true;
157 }
158 return false;
159}

161FetchResult SinucaTraceReader::Fetch(InstructionPacket *ret, int tid) {
162 if (this->HasExecutionEnded()) {
163 return FetchResultEnd;
164 }
165 if (this->threadDataVec[tid]->dynFile.HasReachedEnd()) {
166 return FetchResultNop;
167 }
168 if (this->IsThreadSleeping(tid)) {
169 return FetchResultNop;
170 }
171
172 /* Detect need to fetch new basic block */
173 if (!this->threadDataVec[tid]->isInsideBasicBlock) {
174 this->threadDataVec[tid]->currentInst = 0;
175 if (this->FetchBasicBlock(tid)) {
176 if (this->fetchFailed) {
177 return FetchResultError;
178 } else {
179 return FetchResultNop;
180 }
181 }
182 this->threadDataVec[tid]->isInsideBasicBlock = true;
183 }
184
185 this->ResetInstructionPacket(ret);
186
187 ret->staticInfo =
188 &this->instructionDict[this->threadDataVec[tid]->currentBasicBlock]
189 [this->threadDataVec[tid]->currentInst];
190 if (ret->staticInfo->instReadsMemory || ret->staticInfo->instWritesMemory) {
191 if (this->FetchMemoryData(ret, tid)) {
192 return FetchResultError;
193 }
194 }
195
196 ++this->threadDataVec[tid]->currentInst;
197 if (this->threadDataVec[tid]->currentInst >=
198 this->basicBlockSizeArr[this->threadDataVec[tid]->currentBasicBlock]) {
199 this->threadDataVec[tid]->isInsideBasicBlock = false;
200 }
201
202 this->threadDataVec[tid]->fetchedInst++;
203
204 return FetchResultOk;
205}

223int SinucaTraceReader::FetchBasicBlock(int tid) {
224 if (this->threadDataVec[tid]->dynFile.ReadDynamicRecord()) {
225 if (this->threadDataVec[tid]->dynFile.HasReachedEnd()) {
226 SINUCA3_DEBUG_PRINTF("[FetchBasicBlock] thread [%u] file reached "
227 "end!\n", tid);
228 } else {
229 this->fetchFailed = true;
230 }
231 return 1;
232 }
233
234 static int criticalCont = 0;
235 static int barrierCont = 0;
236
237 DynamicTraceRecordType recType = this->threadDataVec[tid]->dynFile
238 .GetRecordType();
239
240 while (recType == DynamicRecordThreadEvent) {
241 ThreadEventType evType = this->threadDataVec[tid]->dynFile
242 .GetThreadEvent();
243
244 SINUCA3_DEBUG_PRINTF("[FetchBasicBlock] Fetched thread event [%u] in "
245 "thread [%d]\n", evType, tid);
246
247 if (evType == ThreadEventAbruptEnd) {
248 this->reachedAbruptEnd = true;
249 SINUCA3_WARNING_PRINTF(
250 "Trace reader fetched abrupt end event in thread [%d]!\n", tid);
251 return 1; // no basic block to fetch
252 } else if (evType == ThreadEventCriticalStart) {
253 criticalCont++;
254 SINUCA3_DEBUG_PRINTF("Critical region found in thread [%u] and "
255 "criticalCont is [%d]\n", tid, criticalCont);
256 for (int i = 0; i < this->totalThreads; i++) {
257 if (i == tid) continue;
258 this->threadDataVec[i]->isThreadAwake = false;
259 }
260
261 } else if (evType == ThreadEventCriticalEnd) {
262 criticalCont--;
263 if (criticalCont == 0) {
264 SINUCA3_DEBUG_PRINTF("End of critical region. Waking up all "
265 "threads!\n");
266 for (int i = 0; i < this->totalThreads; i++) {
267 this->threadDataVec[i]->isThreadAwake = true;
268 }
269 } else if (criticalCont < 0) {
270 SINUCA3_ERROR_PRINTF("[FetchBasicBlock] criticalCont is "
271 "negative!\n");
272 this->fetchFailed = true;
273 }
274 } else if (evType == ThreadEventBarrierSync) {
275 barrierCont++;
276 if (barrierCont == this->totalThreads) {
277 SINUCA3_DEBUG_PRINTF("[FetchBasicBlock] Threads reached barrier"
278 " sync. Waking up all threads!\n");
279 for (int i = 0; i < this->totalThreads; i++) {
280 this->threadDataVec[i]->isThreadAwake = true;
281 }
282 barrierCont = 0;
283 } else {
284 this->threadDataVec[tid]->isThreadAwake = false;
285 return 1; // no basic block to fetch
286 }
287 } else {
288 SINUCA3_ERROR_PRINTF("[FetchBasicBlock] Unkown thread event [%d]!\n",
289 evType);
290 return 1;
291 }
292
293 if (this->threadDataVec[tid]->dynFile.ReadDynamicRecord()) {
294 if (this->threadDataVec[tid]->dynFile.HasReachedEnd()) {
295 SINUCA3_DEBUG_PRINTF("[FetchBasicBlock] thread [%u] file "
296 "reached end!\n", tid);
297 } else {
298 this->fetchFailed = true;
299 }
300 return 1;
301 }
302 recType = this->threadDataVec[tid]->dynFile.GetRecordType();
303 }
304
305 if (recType != DynamicRecordBasicBlockIdentifier) {
306 SINUCA3_ERROR_PRINTF("[FetchBasicBlock] not expected rec type [%u]\n",
307 recType);
308 this->fetchFailed = true;
309 return 1;
310 }
311
312 unsigned int bblIndex =
313 this->threadDataVec[tid]->dynFile.GetBasicBlockIdentifier();
314 this->threadDataVec[tid]->currentBasicBlock = bblIndex;
315
316 SINUCA3_DEBUG_PRINTF("Bbl fetched is [%d] and it has [%d] inst\n", bblIndex,
317 this->basicBlockSizeArr[bblIndex]);
318
319 return 0;
320}
