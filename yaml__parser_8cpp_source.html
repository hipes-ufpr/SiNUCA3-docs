34static inline yaml::YamlLocation LocationFromMark(const yaml_mark_t mark,
35 const char* file) {
36 yaml::YamlLocation location;
37 location.column = mark.column;
38 location.line = mark.line;
39 location.file = file;

53int yaml::Parser::EnsureFileIsYamlMapping(YamlLocation* location,
54 const char* file) {
55 yaml_event_t event;
56
57 // Assert toplevel is sane: stream start, then document, then mapping.
58 if (this->YamlParseAndLogError(&event)) return 1;
59 if (event.type != YAML_STREAM_START_EVENT) {
60 SINUCA3_ERROR_PRINTF(
61 "while reading config file %s: file is not a YAML mappnig.\n",
62 file);
63 yaml_event_delete(&event);
64 return 1;
65 }
66 yaml_event_delete(&event);
67 if (this->YamlParseAndLogError(&event)) return 1;
68 if (event.type != YAML_DOCUMENT_START_EVENT) {
69 SINUCA3_ERROR_PRINTF(
70 "while reading config file %s: file is not a YAML mappnig.\n",
71 file);
72 yaml_event_delete(&event);
73 return 1;
74 }
75 yaml_event_delete(&event);
76 if (this->YamlParseAndLogError(&event)) return 1;
77 if (event.type != YAML_MAPPING_START_EVENT) {
78 SINUCA3_ERROR_PRINTF(
79 "while reading config file %s: file is not a YAML mappnig.\n",
80 file);
81 yaml_event_delete(&event);
82 return 1;
83 }
84
85 // Write the location.
86 *location = LocationFromMark(event.start_mark, file);
87
88 yaml_event_delete(&event);
89 return 0;
90}

92int yaml::Parser::ParseMapping(const YamlLocation location, const char* anchor,
93 YamlValue* ret) {
94 if (anchor != NULL) {
95 long anchorSize = strlen(anchor);
96 const char* newAnchor = (const char*)this->arena.Alloc(anchorSize + 1);
97 memcpy((void*)newAnchor, (const void*)anchor, anchorSize + 1);
98 anchor = newAnchor;
99 }
100
101 *ret = yaml::YamlValue(yaml::YamlValueTypeMapping, location, anchor);
102 ret->value.mapping =
103 (Map<YamlValue>*)this->arena.Alloc(sizeof(*ret->value.mapping));
104 *ret->value.mapping = Map<YamlValue>(&this->arena);
105
106 yaml_event_t event;
107 yaml_event_type_t eventType;
108 do {
109 if (this->YamlParseAndLogError(&event)) return 1;
110
111 eventType = event.type;
112 if (eventType == YAML_SCALAR_EVENT) {
113 const char* key = (const char*)event.data.scalar.value;
114 yaml::YamlValue value;
115 if (this->ParseYamlValue(location.file, &value)) return 1;
116 ret->value.mapping->Insert(key, value);
117 }
118#ifndef NDEBUG
119 else if (eventType != YAML_MAPPING_END_EVENT) {
120 SINUCA3_DEBUG_PRINTF(
121 "%s:%d: Mapping parser got a strange event: %d\n",
122 __FILE_NAME__, __LINE__, eventType);
123 }
124#endif
125 yaml_event_delete(&event);
126 } while (eventType != YAML_MAPPING_END_EVENT);
127
128 return 0;
129}

131int yaml::Parser::ParseSequence(YamlLocation location, const char* anchor,
132 YamlValue* ret) {
133 if (anchor != NULL) {
134 long anchorSize = strlen(anchor);
135 char* newAnchor = (char*)this->arena.Alloc(anchorSize + 1);
136 memcpy((void*)newAnchor, (const void*)anchor, anchorSize + 1);
137 anchor = newAnchor;
138 }
139
140 *ret = yaml::YamlValue(yaml::YamlValueTypeArray, location, anchor);
141
142 yaml_event_t event;
143 std::vector<yaml::YamlValue> array;
144
145 // The loop breaks when event.type is YAML_SEQUENCE_EVENT.
146 for (;;) {
147 if (this->YamlParseAndLogError(&event)) return 1;
148
149 // Exits by returning the array.
150 if (event.type == YAML_SEQUENCE_END_EVENT) {
151 yaml_event_delete(&event);
152 break;
153 }
154
155 yaml::YamlValue value;
156 int result =
157 this->ParseYamlValueFromEvent(&event, location.file, &value);
158 yaml_event_delete(&event);
159 if (result) return 1;
160 array.push_back(value);
161 }
162
163 ret->value.array.size = array.size();
164 ret->value.array.elements = (YamlValue*)this->arena.Alloc(
165 ret->value.array.size * sizeof(YamlValue));
166 memcpy(ret->value.array.elements, array.data(),
167 array.size() * sizeof(*ret->value.array.elements));
168
169 return 0;
170}

172int yaml::Parser::ParseYamlValueFromEvent(yaml_event_t* event, const char* file,
173 YamlValue* ret) {
174 long strSize;
175 int result = 0;
176 yaml::YamlLocation location = LocationFromMark(event->start_mark, file);
177 switch (event->type) {
178 case YAML_ALIAS_EVENT:
179 *ret = yaml::YamlValue(yaml::YamlValueTypeAlias, location);
180 strSize = strlen((const char*)event->data.alias.anchor);
181 ret->value.alias = (const char*)this->arena.Alloc(strSize + 1);
182 memcpy((void*)ret->value.alias,
183 (const void*)event->data.alias.anchor, strSize + 1);
184 break;
185 case YAML_SCALAR_EVENT:
186 *ret = yaml::YamlValue(yaml::YamlValueTypeString, location,
187 (const char*)event->data.scalar.anchor);
188 strSize = strlen((char*)event->data.scalar.value);
189 ret->value.string = (const char*)this->arena.Alloc(strSize + 1);
190 memcpy((void*)ret->value.string, event->data.scalar.value,
191 strSize + 1);
192 break;
193 case YAML_MAPPING_START_EVENT:
194 result = this->ParseMapping(
195 location, (const char*)event->data.mapping_start.anchor, ret);
196 break;
197 case YAML_SEQUENCE_START_EVENT:
198 result = this->ParseSequence(
199 location, (const char*)event->data.mapping_start.anchor, ret);
200 break;
201 default:
202 SINUCA3_DEBUG_PRINTF(
203 "%s:%d: YamlValue parser got a strange event: %d\n",
204 __FILE_NAME__, __LINE__, event->type);
205 break;
206 }
207
208 return result;
209}

216 yaml_event_delete(&event);

221int yaml::Parser::ParseFile(const char* configFile, YamlValue* const ret) {
222 FILE* fp = fopen(configFile, "r");
223 if (fp == NULL) {
224 SINUCA3_ERROR_PRINTF("No such config file: %s.\n", configFile);
225 return 1;
226 }
227
228 // This only fails on allocation failure, and we don't care about them lmao.
229 yaml_parser_initialize(&this->parser);
230 yaml_parser_set_input_file(&this->parser, fp);
231
232 // We need to make sure the top level is a mapping because another thing
233 // would make no sense.
234 yaml::YamlLocation location;
235 int result = 1;
236 if (!this->EnsureFileIsYamlMapping(&location, configFile))
237 result = this->ParseMapping(location, NULL, ret);
238
239 yaml_parser_delete(&this->parser);
240
241 fclose(fp);
242 return result;
243}

245int yaml::Parser::ParseString(const char* const string, YamlValue* const ret) {
246 yaml_parser_initialize(&this->parser);
247 yaml_parser_set_input_string(&this->parser, (const unsigned char*)string,
248 strlen(string));
249
250 yaml::YamlLocation location;
251 int result = 1;
252 if (!this->EnsureFileIsYamlMapping(&location, "<input string>"))
253 result = this->ParseMapping(location, NULL, ret);
254
255 yaml_parser_delete(&this->parser);
256 return result;
257}

259int yaml::Parser::IncludeString(Map<YamlValue>* config, const char* string) {
260 Parser newParser;
261 yaml::YamlValue newFileValue;
262 if (newParser.ParseFile(string, &newFileValue)) return 1;
263
264 assert(newFileValue.type == yaml::YamlValueTypeMapping);
265 Map<YamlValue>* newValues = newFileValue.value.mapping;
266
267 if (this->ProcessIncludeEntries(&newFileValue)) return 1;
268
269 newValues->ResetIterator();
270 YamlValue newValue;
271 for (const char* key = newValues->Next(&newValue); key != NULL;
272 key = newValues->Next(&newValue)) {
273 config->Insert(key, newValue);
274 }
275
276 return 0;
277}

279int yaml::Parser::IncludeArray(Map<YamlValue>* config, YamlArray array,
280 YamlLocation location) {
281 for (unsigned int i = 0; i < array.size; ++i) {
282 const yaml::YamlValue* const value = &array.elements[i];
283 if (value->type != yaml::YamlValueTypeString) {
284 SINUCA3_ERROR_PRINTF(
285 "%s:%lu:%lu: include array members "
286 "should all be string. Got %s at %s:%lu:%lu.\n",
287 location.file, location.line, location.column,
288 value->TypeAsString(), value->location.file,
289 value->location.line, value->location.column);
290 return 1;
291 }
292
293 if (this->IncludeString(config, value->value.string)) return 1;
294 }
295
296 return 0;
297}

299int yaml::Parser::ProcessIncludeEntries(yaml::YamlValue* config) {
300 assert(config->type == yaml::YamlValueTypeMapping);
301
302 Map<YamlValue>* configMapping = config->value.mapping;
303 YamlValue* entry = configMapping->Get("include");
304 if (entry == NULL) return 0;
305
306 switch (entry->type) {
307 case yaml::YamlValueTypeString:
308 if (this->IncludeString(configMapping, entry->value.string)) {
309 SINUCA3_ERROR_PRINTF("%s:%lu:%lu.\n", config->location.file,
310 config->location.line,
311 config->location.column);
312 return 1;
313 }
314 break;
315 case yaml::YamlValueTypeArray:
316 if (this->IncludeArray(configMapping, entry->value.array,
317 config->location)) {
318 SINUCA3_ERROR_PRINTF("%s:%lu:%lu.\n", config->location.file,
319 config->location.line,
320 config->location.column);
321 return 1;
322 }
323 break;
324 default:
325 SINUCA3_ERROR_PRINTF(
326 "%s:%lu:%lu: include should "
327 "be a string or an array of strings.\n",
328 config->location.file, config->location.line,
329 config->location.column);
330 return 1;
331 }
332
333 return 0;
334}
