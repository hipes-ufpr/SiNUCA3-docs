106template <typename Type>
107int DelayQueue<Type>::Enqueue(Input* elem) {
108 if (this->IsFull()) {
109 return 1;
110 }
111 elem->SetRemoval(this->cyclesClock + this->delay);
112 if (this->IsEmpty()) {
113 this->queueFirst = *elem;
114 } else {
115 this->delayBuffer.Enqueue(elem);
116 }
117 this->occupation++;
118 return 0;
119}

121template <typename Type>
122int DelayQueue<Type>::Dequeue(Input* elem) {
123 if (this->IsEmpty()) {
124 return 1;
125 }
126 if (this->queueFirst.removeAt > this->cyclesClock) {
127 return 1; // Not yet time to remove
128 }
129 *elem = this->queueFirst;
130 if (this->occupation > 1) {
131 this->delayBuffer.Dequeue(&this->queueFirst);
132 }
133 this->occupation--;
134 return 0;
135}

137template <typename Type>
138int DelayQueue<Type>::Configure(Config config) {
139 long delay = 0;
140 if (config.Integer("delay", &delay)) return 1;
141 if (delay < 0) return config.Error("delay", "is not >= 0.");
142 this->delay = delay;
143
144 long throughput = 0;
145 if (config.Integer("throughput", &throughput, true)) return 1;
146 if (throughput < 0) return config.Error("throughput", "is not >= 0.");
147 this->throughput = throughput;
148
149 if (config.ComponentReference("sendTo", &this->sendTo, true)) return 1;
150
151 this->sendToId = this->sendTo->Connect(this->throughput);
152
153 if (this->UseDelayBuffer()) {
154 this->SetDelayBufferSize();
155 this->delayBuffer.Allocate(this->delayBufferSize, sizeof(Input));
156 }
157
158 return 0;
159}

161template <typename Type>
162void DelayQueue<Type>::Clock() {
163 this->cyclesClock++;
164
165 Input input;
166 if (this->UseDelayBuffer()) {
167 while (this->Dequeue(&input) == 0) {
168 this->sendTo->SendRequest(sendToId, &input.elem);
169 }
170 }
171 long totalConnections = this->GetNumberOfConnections();
172 for (long i = 0; i < totalConnections; i++) {
173 while (this->ReceiveRequestFromConnection(i, &input.elem) == 0) {
174 if (this->UseDelayBuffer()) {
175 if (this->Enqueue(&input)) return;
176 } else {
177 if (this->sendTo->SendRequest(sendToId, &input.elem)) return;
178 }
179 }
180 }
181}
