141Linkable* Config::GetComponentByMapping(Map<yaml::YamlValue>* config,
142 yaml::YamlLocation location) {
143 yaml::YamlValue* clazzYaml = config->Get("class");
144
145 if (clazzYaml == NULL) {
146 SINUCA3_ERROR_PRINTF("%s:%lu:%lu Component class not passed.\n",
147 location.file, location.line, location.column);
148 return NULL;
149 }
150 if (clazzYaml->type != yaml::YamlValueTypeString) {
151 SINUCA3_ERROR_PRINTF("%s:%lu:%lu Component class is not a string.\n",
152 clazzYaml->location.file, clazzYaml->location.line,
153 clazzYaml->location.column);
154 return NULL;
155 }
156 const char* clazz = clazzYaml->value.string;
157
158 Linkable* component = CreateComponentByClass(clazz);
159 if (component == NULL) {
160 SINUCA3_ERROR_PRINTF("%s:%lu:%lu Component class %s doesn't exists.\n",
161 clazzYaml->location.file, clazzYaml->location.line,
162 clazzYaml->location.column, clazz);
163 return NULL;
164 }
165
166 this->components->push_back(component);
167 if (component->Configure(Config(this->components, this->aliases,
168 this->definitions, config, location)))
169 return NULL;
170
171 return component;
172}
