34int DynamicTraceReader::OpenFile(const char *sourceDir, const char *imageName,
35 int tid) {
36 unsigned long bufferSize;
37 char *path;
38
39 bufferSize = GetPathTidInSize(sourceDir, "dynamic", imageName);
40 path = (char *)alloca(bufferSize);
41 FormatPathTidIn(path, sourceDir, "dynamic", imageName, tid, bufferSize);
42 this->file = fopen(path, "rb");
43 if (this->file == NULL) {
44 SINUCA3_ERROR_PRINTF("Failed to open dynamic trace file!\n");
45 return 1;
46 }
47 if (this->header.LoadHeader(this->file)) {
48 SINUCA3_ERROR_PRINTF("Failed to read dynamic trace header!\n");
49 return 1;
50 }
51
52 return 0;
53}

55int DynamicTraceReader::ReadDynamicRecord() {
56 if (this->reachedEnd) {
57 SINUCA3_ERROR_PRINTF(
58 "[ReadDynamicRecord] already reached end in dynamic trace file!\n");
59 return 1;
60 }
61
62 this->recordArrayIndex =
63 (this->numberOfRecordsRead == 0) ? 0 : this->recordArrayIndex + 1;
64
65 if (this->recordArrayIndex == this->numberOfRecordsRead) {
66 if (this->LoadRecordArray()) {
67 this->reachedEnd = true;
68 return 1;
69 }
70 }
71
72 return 0;
73}

75int DynamicTraceReader::LoadRecordArray() {
76 if (this->file == NULL) return 1;
77
78 this->recordArrayIndex = 0;
79 unsigned long readBytes = 0;
80 readBytes =
81 fread(this->recordArray, 1, sizeof(this->recordArray), this->file);
82
83 this->numberOfRecordsRead = readBytes / sizeof(*this->recordArray);
84
85 return (this->numberOfRecordsRead == 0);
86}
