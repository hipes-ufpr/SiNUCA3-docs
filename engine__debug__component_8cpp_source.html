30int EngineDebugComponent::Configure(Config config) {
31 bool failNow = false;
32 if (config.Bool("failNow", &failNow)) return 1;
33 if (failNow) return config.Error("failNow", "it's true");
34 if (config.ComponentReference("pointerOther", &this->other)) return 1;
35 if (config.ComponentReference("fetch", &this->fetch)) return 1;
36
37 if (this->other != NULL) this->otherConnectionID = this->other->Connect(0);
38 if (this->fetch != NULL) this->fetchConnectionID = this->fetch->Connect(0);
39
40 return 0;
41}

43void EngineDebugComponent::Clock() {
44 SINUCA3_DEBUG_PRINTF("%p: Clock!\n", this);
45
46 if (this->fetch != NULL) {
47 FetchPacket packet;
48 packet.request = 0;
49 SINUCA3_DEBUG_PRINTF("%p: Fetching instruction!\n", this);
50 this->fetch->SendRequest(this->fetchConnectionID, &packet);
51 if (this->fetch->ReceiveResponse(this->fetchConnectionID, &packet) ==
52 0) {
53 SINUCA3_DEBUG_PRINTF("%p: Received instruction %s\n", this,
54 packet.response.staticInfo->instMnemonic);
55 }
56 }
57
58 InstructionPacket messageInput = {NULL, DynamicInstructionInfo(), 0};
59 InstructionPacket messageOutput = {NULL, DynamicInstructionInfo(), 0};
60
61 if (this->other) {
62 if (!(this->send)) {
63 messageInput.staticInfo = (const StaticInstructionInfo*)0xcafebabe;
64 SINUCA3_DEBUG_PRINTF("%p: Sending message (%p) to %p.\n", this,
65 messageInput.staticInfo, this->other);
66 this->other->SendRequest(this->otherConnectionID, &messageInput);
67 this->send = true;
68 } else {
69 if (this->other->ReceiveResponse(this->otherConnectionID,
70 &messageOutput) == 0) {
71 SINUCA3_DEBUG_PRINTF("%p: Received response (%p) from %p.\n",
72 this, messageOutput.staticInfo,
73 this->other);
74 this->send = false;
75 } else {
76 SINUCA3_DEBUG_PRINTF("%p: No response from %p.\n", this,
77 this->other);
78 }
79 }
80 } else {
81 for (long i = 0; i < this->GetNumberOfConnections(); ++i) {
82 if (this->ReceiveRequestFromConnection(i, &messageOutput) == 0) {
83 SINUCA3_DEBUG_PRINTF("%p: Received message (%p)\n", this,
84 messageOutput.staticInfo);
85 messageInput.staticInfo = messageOutput.staticInfo + 1;
86 SINUCA3_DEBUG_PRINTF("%p: Sending response (%p)\n", this,
87 messageInput.staticInfo);
88 this->SendResponseToConnection(i, &messageInput);
89 }
90 }
91 }
92}
