31int iTLB::Configure(Config config) {
32 const char* tempStr_Policy = "lru"; // lru is default
33
34 if (config.Integer("entries", (long*)&this->entries, true)) return 1;
35 if (config.Integer("associativity", (long*)&this->numWays, true)) return 1;
36 if (config.Integer("missPenalty", (long*)&this->missPenalty, true))
37 return 1;
38
39 // Optional
40 if (config.String("policy", &tempStr_Policy)) return 1;
41 if (config.Integer("pageSize", (long*)&this->pageSize)) return 1;
42
43 if (this->entries == 0) {
44 SINUCA3_ERROR_PRINTF(
45 "Invalid value for iTLB parameter \"entries\": should be > "
46 "0.");
47 return 1;
48 }
49
50 if (this->numWays == 0) {
51 SINUCA3_ERROR_PRINTF(
52 "Invalid value for iTLB parameter \"associativity\": should "
53 "be > 0.");
54 return 1;
55 }
56
57 unsigned int numSets = this->entries / this->numWays;
58 this->cache = CacheMemory<unsigned long>::fromNumSets(
59 numSets, this->pageSize, this->numWays, tempStr_Policy);
60 if (this->cache == NULL) {
61 SINUCA3_ERROR_PRINTF("iTLB: Failed to alocate CacheMemory\n");
62 return 1;
63 }
64
65 this->pendingRequests.Allocate(0, sizeof(struct TLBRequest));
66
67 return 0;
68}

70void iTLB::Clock() {
71 SINUCA3_DEBUG_PRINTF("%p: iTLB Clock!\n", this);
72
73 long numberOfConnections = this->GetNumberOfConnections();
74 for (int id = 0; id < numberOfConnections; ++id) {
75 struct TLBRequest newRequest;
76 if (this->ReceiveRequestFromConnection(id, &newRequest.addr) == 0) {
77 ++this->numberOfRequests;
78 newRequest.id = id;
79 this->pendingRequests.Enqueue(&newRequest);
80 SINUCA3_DEBUG_PRINTF("%p: iTLB Message (%p) Received!\n", this,
81 (void*)newRequest.addr);
82 }
83 }
84
85 /* If paying a miss penalty */
86 if (this->currentPenalty > 0) {
87 --this->currentPenalty;
88
89 if (this->currentPenalty == 0) {
90 this->currentPenalty = NO_PENALTY;
91 SINUCA3_DEBUG_PRINTF("%p: iTLB Waiting ended! Sending response\n",
92 this);
93 this->SendResponseToConnection(this->curRequest.id,
94 &this->curRequest.addr);
95 }
96
97 return;
98 }
99
100 struct TLBRequest request;
101 if (this->pendingRequests.Dequeue(&request)) {
102 SINUCA3_DEBUG_PRINTF("%p: iTLB No work avaliable: Stall.\n", this);
103 return;
104 }
105 this->curRequest = request;
106
107 // We dont have (and dont need) data to send back, so
108 // the same address is send back to to signal
109 // that the iTLB's operation has been completed.
110
111 // Read() returns NULL if it was a miss.
112 if (this->cache->Read(this->curRequest.addr)) {
113 SINUCA3_DEBUG_PRINTF("%p: iTLB (%p) HIT Sending response!\n", this,
114 (void*)this->curRequest.addr);
115 this->SendResponseToConnection(this->curRequest.id,
116 &this->curRequest.addr);
117 } else {
118 SINUCA3_DEBUG_PRINTF("%p: iTLB (%p) MISS Waiting %lu cycles!\n", this,
119 (void*)this->curRequest.addr, this->missPenalty);
120 this->currentPenalty = this->missPenalty;
121 this->cache->Write(this->curRequest.addr, &this->curRequest.addr);
122 }
123
124 return;
125}
