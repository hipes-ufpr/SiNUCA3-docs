32int StaticTraceReader::OpenFile(const char *sourceDir, const char *imgName) {
33 unsigned long bufferLen;
34 char *staticPath;
35
36 bufferLen = GetPathTidOutSize(sourceDir, "static", imgName);
37 staticPath = (char *)alloca(bufferLen);
38 FormatPathTidOut(staticPath, sourceDir, "static", imgName, bufferLen);
39
40 /* get file descriptor */
41 this->fileDescriptor = open(staticPath, O_RDONLY);
42 if (fileDescriptor == -1) {
43 printFileErrorLog(staticPath, "O_RDONLY");
44 return 1;
45 }
46 /* get file size */
47 this->mmapSize = lseek(this->fileDescriptor, 0, SEEK_END);
48 /* map file to virtual memory */
49 this->mmapPtr = (char *)mmap(0, this->mmapSize, PROT_READ, MAP_PRIVATE,
50 this->fileDescriptor, 0);
51 if (this->mmapPtr == MAP_FAILED) {
52 printFileErrorLog(staticPath, "PROT_READ MAP_PRIVATE");
53 return 1;
54 }
55
56 this->header.LoadHeader(this->mmapPtr, &this->mmapOffset);
57
58 return 0;
59}

61void *StaticTraceReader::ReadData(unsigned long len) {
62 if (this->mmapOffset >= this->mmapSize) return NULL;
63 void *ptr = (void *)(this->mmapPtr + this->mmapOffset);
64 this->mmapOffset += len;
65 return ptr;
66}

72 SINUCA3_ERROR_PRINTF("Failed to read static trace record\n");

79void StaticTraceReader::TranslateRawInstructionToSinucaInst(
80 StaticInstructionInfo *instInfo) {
81 if (instInfo == NULL) return;
82
83 Instruction *rawInst = &this->record->data.instruction;
84
85 strcpy(instInfo->instMnemonic, rawInst->instructionMnemonic);
86
87 instInfo->instSize = rawInst->instructionSize;
88 instInfo->instAddress = rawInst->instructionAddress;
89 instInfo->instPerformsAtomicUpdate = rawInst->instPerformsAtomicUpdate;
90 instInfo->instCausesCacheLineFlush = rawInst->instCausesCacheLineFlush;
91 instInfo->isPredicatedInst = rawInst->isPredicatedInst;
92 instInfo->instReadsMemory = rawInst->instReadsMemory;
93 instInfo->instWritesMemory = rawInst->instWritesMemory;
94 instInfo->isIndirectControlFlowInst = rawInst->isIndirectCtrlFlowInst;
95 instInfo->numberOfReadRegs = rawInst->rRegsArrayOccupation;
96 instInfo->numberOfWriteRegs = rawInst->wRegsArrayOccupation;
97
98 unsigned long occupationInBytes;
99
100 occupationInBytes =
101 sizeof(*rawInst->readRegsArray) * rawInst->rRegsArrayOccupation;
102 memcpy(instInfo->readRegsArray, rawInst->readRegsArray, occupationInBytes);
103
104 occupationInBytes =
105 sizeof(*rawInst->writtenRegsArray) * rawInst->wRegsArrayOccupation;
106 memcpy(instInfo->writtenRegsArray, rawInst->writtenRegsArray,
107 occupationInBytes);
108
109 if (rawInst->isCallInstruction) {
110 instInfo->branchType = BranchCall;
111 } else if (rawInst->isSyscallInstruction) {
112 instInfo->branchType = BranchSyscall;
113 } else if (rawInst->isRetInstruction) {
114 instInfo->branchType = BranchRet;
115 } else if (rawInst->isSysretInstruction) {
116 instInfo->branchType = BranchSysret;
117 } else if (rawInst->isBranchInstruction) {
118 if (rawInst->instHasFallthrough) {
119 instInfo->branchType = BranchCond;
120 } else {
121 instInfo->branchType = BranchUncond;
122 }
123 }
124
125 this->record = NULL;
126}
